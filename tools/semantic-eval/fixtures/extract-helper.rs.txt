struct Bucket {
    tokens: f64,
    updated: u64,
}

impl Bucket {
    fn refill(&mut self, rate: f64, capacity: f64, now: u64) {
        let elapsed = (now - self.updated) as f64;
        self.tokens = (self.tokens + elapsed * rate).min(capacity);
        self.updated = now;
    }
}

impl Limiter {
    fn allow(&mut self, key: &str, now: u64) -> bool {

    }
